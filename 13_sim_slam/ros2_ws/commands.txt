# Frequently used commands inside the container's browser desktop terminal(s)
# (http://localhost:6080). Not for your host machine.
# Pasted in order for what you should need


# --- Build your package after editing files ---
# Note: you only need to do this once; built files are stored on your computer outside container
cd ~/ros2_ws
colcon build --symlink-install

# --- Launch Gazebo with your custom robot in the brick square world ---
cd ~/ros2_ws
source install/setup.bash
ros2 launch goosebot_0 spawn_robot.launch.py world:=/home/ubuntu/ros2_ws/src/brick_area.world

# --- Run ORB-SLAM3 + octomap (second terminal) ---
cd ~/ros2_ws
source install/setup.bash
ros2 launch orbslam3_ros2 orbslam3_ros2.launch.py camera_type:=mono start_octomap:=true

# --- Run the wall-avoider script (third terminal, after SLAM and simulation are running) ---
cd ~/ros2_ws
source install/setup.bash
ros2 launch goosebot_0 nav2_bringup.launch.py
